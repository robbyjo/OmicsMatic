FORMALS
as.pairlist(alist(Z = , y = , nPermu = 1e+05, epsilon = 10^-6))
BODY
{
    nSNP = ncol(Z)
    nSample = length(y)
    RE <- .C("RSKATPermu", as.double(as.vector(Z)), as.integer(y), as.integer(nSNP), as.integer(nSample), as.integer(nPermu), double(1), double(1), as.double(epsilon))
    pval.re <- cbind(RE[[6]], RE[[7]])
    pval1 <- cbind(pval.re[, 1], pval.re[, 1] - pval.re[, 2]/2, pval.re[, 2])
    re = list(p.value = pval1[1, 2], p.value.standard = pval1[1, 1])
    return(re)
}
